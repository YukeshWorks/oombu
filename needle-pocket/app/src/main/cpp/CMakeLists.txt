cmake_minimum_required(VERSION 3.22.1)
project(needlebridge LANGUAGES C CXX)
set(VENDOR_DIR "${CMAKE_CURRENT_SOURCE_DIR}/vendor/${ANDROID_ABI}")
if(NOT EXISTS "${VENDOR_DIR}/libneedle.a" OR NOT EXISTS "${VENDOR_DIR}/needle.h")
  message(FATAL_ERROR "Official Needle Android engine files are missing")
endif()
add_library(needle_engine STATIC IMPORTED)
set_target_properties(needle_engine PROPERTIES IMPORTED_LOCATION "${VENDOR_DIR}/libneedle.a")
add_library(needlebridge SHARED native-lib.cpp)
target_include_directories(needlebridge PRIVATE "${VENDOR_DIR}")
target_compile_options(needlebridge PRIVATE -Wall -Wextra -Wno-unused-parameter)
target_link_options(needlebridge PRIVATE "-Wl,-z,max-page-size=16384")
find_library(log-lib log)
find_library(android-lib android)
find_library(cxx-lib c++_shared)
target_link_libraries(needlebridge PRIVATE needle_engine ${cxx-lib} ${log-lib} ${android-lib})
